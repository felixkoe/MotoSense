RASPBERRY PI ACCESS POINT + SYSTEMD SERVICE
============================================

SYSTEM
- sudo apt update && sudo apt upgrade -y

ACCESS POINT
- sudo nmcli con add type wifi ifname wlan0 con-name MotoSense-AP ssid "MotoSense" mode ap
- sudo nmcli con modify MotoSense-AP wifi-sec.key-mgmt wpa-psk
- sudo nmcli con modify MotoSense-AP wifi-sec.psk "motosense123"
- sudo nmcli con modify MotoSense-AP ipv4.method shared ipv4.address 192.168.4.1/24
- sudo nmcli con up MotoSense-AP
- Pi ist danach erreichbar unter 192.168.4.1
- Dashboard: http://192.168.4.1:8060

VENV (falls noch nicht vorhanden)
- mkdir ~/motosense && cd ~/motosense
- python3 -m venv venv
- source venv/bin/activate
- pip install paho-mqtt dash dash-leaflet plotly

SYSTEMD SERVICE
- sudo nano /etc/systemd/system/motosense-dashboard.service

  [Unit]
  Description=MotoSense Dashboard
  After=network-online.target mosquitto.service
  Wants=network-online.target

  [Service]
  Type=simple
  User=pi
  WorkingDirectory=/home/pi/motosense
  ExecStart=/home/pi/motosense/venv/bin/python /home/pi/motosense/dashboard.py
  Restart=on-failure
  RestartSec=5

  [Install]
  WantedBy=multi-user.target

- sudo systemctl daemon-reload
- sudo systemctl enable motosense-dashboard
- sudo systemctl start motosense-dashboard
- sudo systemctl status motosense-dashboard  ->  "active (running)" = OK
- sudo reboot  ->  Autostart testen

NUETZLICHE BEFEHLE
- sudo systemctl restart motosense-dashboard
- sudo journalctl -u motosense-dashboard -f
